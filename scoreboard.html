scoreboard.yml

java:
  enabled: true
  update-interval: 20
  show-numbers: false
  title: "<animation:scoreboard-title>"
  lines:
    - "&fPlayer: %player_name%"
    - "&fHealth: %gtabs_health%"
    - "&fPing: %player_ping%"

bedrock:
  enabled: true
  update-interval: 20
  show-numbers: false
  title: "<animation:scoreboard-title>"
  lines:
    - "&f%player_name%"
    - "%gtabs_health%"
    - "&7Online: &f%server_online%"

show-numbers: false requests hidden sidebar numbers on supported modern implementations. Players can use /gsbtoggle.
